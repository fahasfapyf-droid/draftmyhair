━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
PIXIE CHECK
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Verify:

• compact silhouette
• crown carries greatest weight
• sides remain short
• nape remains cropped
• lower perimeter remains light
• nothing approaches jaw length

FAIL if the result reads primarily as a curly bob.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
CURL CHECK
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Verify:

• 2C-to-loose-3A appearance
• mixed curl diameters
• varied curl openness
• varied orientation
• natural curl overlap
• partially relaxed curls
• subtle curl imperfections

FAIL if:

• identical ringlets repeat
• curls appear decorative
• every curl has identical definition
• curls appear artificially generated

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
FRINGE CHECK
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Verify:

• short
• soft
• curly
• irregular
• partially visible forehead

FAIL if:

• heavy fringe
• curtain bangs
• dominant centre curl
• matching forehead ringlets

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
SIDES & NAPE CHECK
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Verify:

• sides close to head
• partial ear visibility
• soft taper
• cropped nape
• no curl accumulation

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
BOB DRIFT CHECK
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

FAIL if:

• lower curl mass becomes heavy
• jawline develops a continuous perimeter
• back becomes grown-out
• curls touch neck or shoulders

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
REALISM CHECK
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Verify:

• natural root emergence
• realistic curl compression
• natural density
• realistic gravity
• source lighting retained
• source grain retained
• source sharpness retained

Reject:

• plastic curls
• wet finish
• crunchy styling
• wig appearance
• AI-smooth hair

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Minimum production threshold:

Overall ≥ 9.4

Identity ≥ 9.5

Pose ≥ 9.5

Geometry ≥ 9.5

Haircut Accuracy ≥ 9.5

Curl Realism ≥ 9.5

Hair Integration ≥ 9.4

If criteria are satisfied, retain v1.
